cmake_minimum_required(VERSION 3.22)
project(dependencies LANGUAGES NONE)

find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
find_program(NINJA_EXECUTABLE ninja REQUIRED)

if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    set(CMAKE_INSTALL_PREFIX "${CMAKE_BINARY_DIR}/install" CACHE PATH "Native install directory" FORCE)
endif()

set(ANDROID_NDK "$ENV{ANDROID_NDK_HOME}" CACHE PATH "Android NDK directory")
if(NOT ANDROID_NDK)
    set(android_sdk "$ENV{ANDROID_HOME}")
    if(NOT android_sdk)
        set(android_sdk "$ENV{ANDROID_SDK_ROOT}")
    endif()
    if(NOT android_sdk)
        if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
            set(android_sdk "$ENV{HOME}/Library/Android/sdk")
        else()
            set(android_sdk "$ENV{HOME}/Android/Sdk")
        endif()
    endif()
    set(ANDROID_NDK "${android_sdk}/ndk/29.0.13846066" CACHE PATH "Android NDK directory" FORCE)
endif()
if(NOT EXISTS "${ANDROID_NDK}/build/cmake/android.toolchain.cmake")
    message(FATAL_ERROR "Android NDK toolchain not found in ${ANDROID_NDK}")
endif()

add_subdirectory(px4)
